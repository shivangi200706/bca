1.CLIENT_MASTER_23 
CREATE TABLE CLIENT_MASTER_23 ( CLIENTNO VARCHAR2(6) PRIMARY KEY CHECK (CLIENTNO LIKE 'C%'), NAME VARCHAR2(20)
 NOT NULL, ADDRESS1 VARCHAR2(30), ADDRESS2 VARCHAR2(30), CITY VARCHAR2(15), PINCODE NUMBER(8)
 , STATE VARCHAR2(15), BALDUE NUMBER(10,2) );
 
2.PRODUCT_MASTER_23 
 CREATE TABLE PRODUCT_MASTER_23 ( PRODUCTNO VARCHAR2(6) PRIMARY KEY CHECK (PRODUCTNO LIKE 'P%'), 
 DESCRIPTION VARCHAR2(15) NOT NULL, PROFITPERCENT NUMBER(4,2) NOT NULL, UNITMASTER VARCHAR2(10) 
 NOT NULL, QTYONHAND NUMBER(8) NOT NULL, REORDERLVL NUMBER(8) NOT NULL, SELLPRICE NUMBER(8,2)
 NOT NULL CHECK (SELLPRICE<>0), COSTPRICE NUMBER(8,2) NOT NULL CHECK (COSTPRICE<>0) );
 
3. SALESMAN_MASTER_23
 CREATE TABLE SALESMAN_MASTER_23 ( SALESMANNO VARCHAR2(6) PRIMARY KEY CHECK (SALESMANNO LIKE 'S%'), 
 SALESMANNAME VARCHAR2(20) NOT NULL, ADDRESS1 VARCHAR2(30) NOT NULL, ADDRESS2 VARCHAR2(30), CITY VARCHAR2(20),
 PINCODE NUMBER(8), STATE VARCHAR2(20), SALAMT NUMBER(8,2) NOT NULL CHECK (SALAMT<>0), TGTTOGET NUMBER(6,2)
 NOT NULL CHECK (TGTTOGET<>0), YTDSALES NUMBER(6,2) NOT NULL CHECK (YTDSALES<>0), REMARKS VARCHAR2(60) ); 
 
4. SALES_ORDER_23 CREATE TABLE SALES_ORDER_23 ( ORDERNO VARCHAR2(6) PRIMARY KEY CHECK (ORDERNO LIKE 'O%'),
 CLIENTNO VARCHAR2(6) REFERENCES CLIENT_MASTER_23(CLIENTNO), ORDERDATE DATE NOT NULL, DELYADDR VARCHAR2(25), 
 SALESMANNO VARCHAR2(6) REFERENCES SALESMAN_MASTER_23(SALESMANNO), DELYTYPE CHAR(1) CHECK (DELYTYPE IN 
 ('P','F')), BIILYN CHAR(1), DELYDATE DATE, ORDERSTATUS VARCHAR2(10) );


5. SALES_ORDER_DETAILS_23 CREATE TABLE SALES_ORDER_DETAILS_23 ( ORDERNO VARCHAR2(6) REFERENCES SALES_ORDER_23
(ORDERNO), PRODUCTNO VARCHAR2(6) REFERENCES PRODUCT_MASTER_23(PRODUCTNO), QTYORDERED NUMBER(8), 
QTYDISP NUMBER(8), PRODUCTRATE NUMBER(10,2) );

SELECT NAME FROM CLIENT_MASTER_23 WHERE LOWER(NAME) LIKE '_a%'; 
SELECT CLIENTNO, NAME, CITY FROM CLIENT_MASTER_23 WHERE CITY LIKE 'M%';
SELECT CLIENTNO, NAME, CITY FROM CLIENT_MASTER_23 WHERE CITY IN ('Bangalore','Mangalore'); 
SELECT CLIENTNO, NAME, BALDUE FROM CLIENT_MASTER_23 WHERE BALDUE > 10000; 
SELECT ORDERNO, CLIENTNO, ORDERDATE FROM SALES_ORDER_22 WHERE EXTRACT(MONTH FROM ORDERDATE)=6; 
SELECT ORDERNO, CLIENTNO, ORDERDATE FROM SALES_ORDER_22 WHERE CLIENTNO IN ('C00001','C00002'); 
SELECT PRODUCT_NO, DESCRIPTION, SELLPRICE FROM PRODUCT_MASTER_23 WHERE SELLPRICE > 500 AND SELLPRICE <= 750;
SELECT PRODUCT_NO, DESCRIPTION, SELLPRICE FROM PRODUCT_MASTER_23 WHERE SELLPRICE > 500; 
SELECT PRODUCT_NO, SELLPRICE, SELLPRICE*0.15 AS NEW_PRICE FROM PRODUCT_MASTER_23; 
SELECT SELLPRICE*0.15 AS NEW_PRICE FROM PRODUCT_MASTER_23;
SELECT NAME, CITY, STATE FROM CLIENT_MASTER_23 WHERE STATE <> 'Maharashtra';
SELECT ROUND(AVG(SELLPRICE),2) AS AVG_PRICE FROM PRODUCT_MASTER_23;   
SELECT MAX(SELLPRICE) AS MAX_PRICE, MIN(SELLPRICE) AS MIN_PRICE FROM PRODUCT_MASTER_23; 
SELECT COUNT(*) AS PRODUCT_COUNT FROM PRODUCT_MASTER_23 WHERE SELLPRICE <= 500;
SELECT PRODUCT_NO, DESCRIPTION, QTYPNHAND , REORDERLVL FROM PRODUCT_MASTER_23 WHERE QTYPNHAND  < REORDERLVL;
